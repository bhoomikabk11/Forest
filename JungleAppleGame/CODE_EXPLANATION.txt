def create_background():
    """Create attractive jungle background with depth and details"""
    
    # STEP 1: Create blank canvas
    img = Image.new('RGB', (800, 600), color='#87CEEB')
    
    EXPLANATION:
    - Image.new() creates a new blank image
    - 'RGB' = Red, Green, Blue color mode (each pixel has 3 color values)
    - (800, 600) = Width x Height in pixels
    - color='#87CEEB' = Sky blue color in hexadecimal format
    
    # STEP 2: Create drawing object
    draw = ImageDraw.Draw(img)
    
    EXPLANATION:
    - ImageDraw.Draw() creates a "pen" to draw on the image
    - Now we can use draw.rectangle(), draw.ellipse(), etc.
    
    # STEP 3: Sky gradient (makes sky look realistic)
    for y in range(0, 350):
        color_val = int(135 + (y / 350) * 50)
        draw.rectangle([0, y, 800, y+1], fill=(135, 206, color_val))
    
    EXPLANATION:
    - Loop through Y positions from 0 to 350 (top to middle)
    - Calculate changing blue value (creates gradient effect)
    - Draw thin horizontal line at each Y position
    - Result: Sky fades from light to lighter blue
    
    FORMULA BREAKDOWN:
    - y = 0: color_val = 135 + 0 = 135 (darker)
    - y = 175: color_val = 135 + 25 = 160 (medium)
    - y = 350: color_val = 135 + 50 = 185 (lighter)
    
    # STEP 4: Mountains (background depth)
    mountain_points = [
        (0, 280), (150, 220), (300, 260), (450, 200), 
        (600, 240), (800, 220), (800, 350), (0, 350)
    ]
    draw.polygon(mountain_points, fill='#4a6741', outline='#3d5535')
    
    EXPLANATION:
    - mountain_points = list of (x, y) coordinates
    - draw.polygon() connects all points to form a shape
    - fill = interior color
    - outline = border color
    - Creates mountain silhouette in background
    
    # STEP 5: Trees with depth
    for x in range(50, 800, 120):  # x = 50, 170, 290, ...
        trunk_h = 200 + (x % 80)
        draw.rectangle([x-8, 600-trunk_h, x+8, 600], fill='#2d1810')
    
    EXPLANATION:
    - range(50, 800, 120) = start at 50, go to 800, step by 120
    - trunk_h = varying height (makes trees different sizes)
    - x % 80 = remainder when dividing x by 80 (creates variation)
    - Rectangle coordinates: [left, top, right, bottom]
    
    # STEP 6: Tree crowns (foliage)
    for offset in [(0, -40), (-15, -25), (15, -25), (0, -10)]:
        cx, cy = x + offset[0], 600 - trunk_h + offset[1]
        draw.ellipse([cx-35, cy-35, cx+35, cy+35], fill='#0d2d0a')
    
    EXPLANATION:
    - offset = positions relative to tree trunk
    - cx, cy = center point for each circle
    - ellipse parameters: [left, top, right, bottom]
    - Multiple circles create bushy tree appearance
    
    # STEP 7: Flowers (decorative elements)
    for i in range(20):  # Create 20 flowers
        fx = 40 + (i * 38)  # Spread horizontally
        fy = 560 + (i % 3) * 15  # Vary vertical position
        f_color = flower_colors[i % 4]  # Cycle through 4 colors

                fx + petal[0] - 3, fy + petal[1] - 3,
                fx + petal[0] + 3, fy + petal[1] + 3
            ], fill=f_color)
    
    EXPLANATION:
    - i % 3 = cycles 0, 1, 2, 0, 1, 2... (creates wave pattern)
    - i % 4 = cycles 0, 1, 2, 3, 0... (4 different flower colors)

    def preprocess_image(self, image_input):
        """
        Convert any image format to standard 64x64 RGB array
        """
        # Handle different input types
        if isinstance(image_input, pygame.Surface):
            # Convert pygame Surface to PIL Image
            width, height = image_input.get_size()
            img_str = pygame.image.tostring(image_input, 'RGB')
            pil_image = Image.frombytes('RGB', (width, height), img_str)

EXPLANATION:
- isinstance() = checks what type of object we have
- pygame.Surface = image format used by Pygame
- pygame.image.tostring() = converts Surface to raw bytes
- Image.frombytes() = creates PIL Image from bytes

WHY MULTIPLE FORMATS?
- Pygame uses pygame.Surface
- PIL/Pillow uses PIL.Image
- NumPy uses arrays
- Need to convert between them!

        elif isinstance(image_input, str):
            # Load from file path
            pil_image = Image.open(image_input).convert('RGB')
        
        elif isinstance(image_input, Image.Image):
            pil_image = image_input.convert('RGB')
        
        elif isinstance(image_input, np.ndarray):
            pil_image = Image.fromarray(image_input.astype('uint8'), 'RGB')

EXPLANATION:
- Handles 4 different input types
- All converted to PIL Image
- .convert('RGB') ensures 3 color channels

        # Resize to standard size
        pil_image = pil_image.resize((64, 64))
        
EXPLANATION:
- CNN expects consistent input size
- All images resized to 64x64 pixels
- Makes processing uniform

        # Convert to numpy array and normalize
        img_array = np.array(pil_image) / 255.0
        
EXPLANATION:
- np.array() converts PIL Image to NumPy array
- Division by 255.0 normalizes values from 0-255 to 0.0-1.0
- Why normalize? Makes math calculations more stable

EXAMPLE:
Before: [255, 128, 0]  (bright red, medium green, no blue)
After:  [1.0, 0.5, 0.0]  (normalized values)

        return img_array

    def classify_apple(self, image_input):
        """
        Classify apple as GOOD or DAMAGED using color analysis
        """
        # STEP 1: Preprocess image
        img_array = self.preprocess_image(image_input)
        
        # STEP 2: Extract color features
        mean_red = np.mean(img_array[:, :, 0])
        mean_green = np.mean(img_array[:, :, 1])
        mean_blue = np.mean(img_array[:, :, 2])

EXPLANATION OF np.mean():
- img_array shape: (64, 64, 3)
  * 64 rows, 64 columns, 3 color channels
- img_array[:, :, 0] = all rows, all columns, channel 0 (RED)
- np.mean() = average of all values
- Result: Average redness of entire image

EXAMPLE:
If apple is very red:
  mean_red = 0.8 (high)
  mean_green = 0.1 (low)
  mean_blue = 0.05 (low)

If apple is brown (damaged):
  mean_red = 0.3 (medium)
  mean_green = 0.2 (medium)
  mean_blue = 0.1 (low)

        # STEP 3: Calculate derived features
        brightness = (mean_red + mean_green + mean_blue) / 3

EXPLANATION:
- Average of all three channels
- Measures overall lightness
- Bright apple = high brightness
- Dark/rotten apple = low brightness

        red_dominance = mean_red - (mean_green + mean_blue) / 2

EXPLANATION:
- How much red stands out from other colors
- Good apple: Red >> Green + Blue → high red_dominance
- Bad apple: Colors more balanced → low red_dominance

MATH EXAMPLE:
Good apple: 0.8 - (0.1 + 0.05)/2 = 0.8 - 0.075 = 0.725 (HIGH!)
Bad apple:  0.3 - (0.2 + 0.1)/2 = 0.3 - 0.15 = 0.15 (LOW)

        color_balance = abs(mean_red - mean_green)

EXPLANATION:
- Difference between red and green
- abs() = absolute value (always positive)
- High value = distinct color
- Low value = muddy/brown color

        # STEP 4: Classification decision
        if mean_red > 0.25 and red_dominance > 0.25:
            prediction = 'GOOD'
            confidence = min(0.99, 0.80 + red_dominance)
        else:
            prediction = 'DAMAGED'
            confidence = min(0.99, 0.75 + abs(0.2 - mean_red))

EXPLANATION OF LOGIC:
- IF red is strong (>0.25) AND red dominates (>0.25)
  THEN apple is GOOD
- ELSE apple is DAMAGED

CONFIDENCE CALCULATION:
- min(0.99, value) = never exceed 99%
- 0.80 + red_dominance = base 80% + extra based on how red
- Higher red dominance = higher confidence

EXAMPLE:
red_dominance = 0.15 → confidence = min(0.99, 0.80 + 0.15) = 0.95 = 95%
red_dominance = 0.30 → confidence = min(0.99, 0.80 + 0.30) = 0.99 = 99%

        return prediction, confidence

RETURNS:
- prediction: String 'GOOD' or 'DAMAGED'
- confidence: Float 0.0 to 0.99 (percentage)

HOW COLLISION WORKS:
- Compares player.rect and apple.rect
- If rectangles overlap → collision!
- Very fast rectangle intersection test

        for apple in hits:
            self.total_collected += 1

EXPLANATION:
- Loop through each apple we hit
- Increment total counter

            # Pass apple image to CNN for classification
            prediction, confidence = self.classifier.classify_apple(apple.image)

🧠 CNN INTEGRATION - THE MAGIC LINE!
- apple.image = pygame Surface with apple graphics
- Sends to our CNN classifier
- Gets back prediction ('GOOD'/'DAMAGED') and confidence (0-1)

            if prediction == 'GOOD':
                self.good_apples += 1
                self.score += 1
                self.message = f"✓ GOOD APPLE! +1 (CNN: {confidence:.0%} confident)"
                self.message_color = GREEN
                print(f"✅ GOOD apple collected! Score: {self.score}")

EXPLANATION:
- Increment good apple count and score
- Create success message
- {confidence:.0%} formats as percentage (0.95 → 95%)
- Print to console for debugging

            else:
                self.damaged_apples += 1
                self.message = f"✗ DAMAGED APPLE - Not good! (CNN: {confidence:.0%})"
                self.message_color = RED
                print(f"❌ DAMAGED apple detected! No points.")

EXPLANATION:
- Increment damaged counter (but not score!)
- Show rejection message in red
- Log to console

            self.message_timer = 120  # Show for 2 seconds

EXPLANATION:
- 120 frames ÷ 60 FPS = 2 seconds
- Message will display for 2 seconds then disappear

--------------------------------------------------------------------------------
METHOD: update() - GAME LOOP LOGIC
--------------------------------------------------------------------------------

    def update(self):
        """Update game state every frame"""
        # Get keyboard input
        keys = pygame.key.get_pressed()

EXPLANATION:
- Returns dictionary of all keys
- keys[pygame.K_LEFT] = True if left arrow pressed
- Checked every frame for smooth movement

        # Update player
        self.player.update(keys)

EXPLANATION:
- Calls player's update method
- Moves player based on keys pressed

        # Update apples
        self.apples.update()

EXPLANATION:
- Calls update() on every apple in group
- Makes apples fall

        # Remove apples that fell off screen
        for apple in self.apples:
            if apple.rect.top > SCREEN_HEIGHT:
                apple.kill()

EXPLANATION:
- apple.rect.top > 600 = apple completely below screen
- .kill() removes sprite from all groups
- Prevents memory leak from infinite apples

        # Spawn new apples
        self.frame_count += 1
        if self.frame_count % APPLE_SPAWN_RATE == 0:
            self.spawn_apple()

EXPLANATION:
- frame_count tracks frames passed
- % = modulo (remainder)
- Every 60 frames (1 second at 60 FPS), spawn apple

EXAMPLE:
frame_count = 59: 59 % 60 = 59 (not 0, don't spawn)
frame_count = 60: 60 % 60 = 0 (spawn!)
frame_count = 61: 61 % 60 = 1 (not 0, don't spawn)

        # Check collisions
        self.check_collisions()
        
        # Update message timer
        if self.message_timer > 0:
            self.message_timer -= 1

EXPLANATION:
- Countdown timer
- When reaches 0, message stops displaying

--------------------------------------------------------------------------------
METHOD: draw() - RENDERING GRAPHICS
--------------------------------------------------------------------------------

    def draw(self):
        """Draw everything on screen"""
        # Draw background
        self.screen.blit(self.background, (0, 0))

EXPLANATION:
- Draws background at position (0, 0)
- Covers entire screen

        # Draw apples
        self.apples.draw(self.screen)

EXPLANATION:
- Group.draw() draws all sprites in group
- One line draws all apples!

        # Draw player
        self.player.draw(self.screen)

        # Draw UI text
        score_text = self.font_large.render(f"Score: {self.score}", True, YELLOW)
        self.screen.blit(score_text, (10, 10))

EXPLANATION:
- .render() creates text surface
- f"Score: {self.score}" = formatted string
- True = anti-aliasing (smooth text)
- YELLOW = text color
- (10, 10) = position

        # Draw message if active
        if self.message_timer > 0:
            msg_text = self.font_medium.render(self.message, True, self.message_color)
            msg_rect = msg_text.get_rect(center=(SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2))

EXPLANATION:
- Only draw if timer > 0
- center=(400, 300) = centered on screen

            # Semi-transparent background
            bg_rect = msg_rect.inflate(20, 10)
            bg_surface = pygame.Surface((bg_rect.width, bg_rect.height))
            bg_surface.set_alpha(200)
            bg_surface.fill(BLACK)
            self.screen.blit(bg_surface, bg_rect)

EXPLANATION:
- .inflate(20, 10) = make rect bigger (padding)
- .set_alpha(200) = semi-transparent (0=invisible, 255=opaque)
- Creates dark background behind text for readability

            self.screen.blit(msg_text, msg_rect)

        # Update display
        pygame.display.flip()

EXPLANATION:
- .flip() shows everything we drew
- Double-buffering: draw to hidden buffer, then flip to screen
- Prevents flickering

--------------------------------------------------------------------------------
METHOD: handle_events() - INPUT HANDLING
--------------------------------------------------------------------------------

    def handle_events(self):
        """Handle game events"""
        for event in pygame.event.get():

EXPLANATION:
- pygame.event.get() = list of events since last check
- Events: mouse clicks, key presses, window close, etc.

            if event.type == pygame.QUIT:
                self.running = False

EXPLANATION:
- QUIT event = user clicked X button
- Sets running to False, which exits game loop

            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    self.running = False

EXPLANATION:
- KEYDOWN = key pressed event
- K_ESCAPE = Escape key
- Allows quitting with ESC

--------------------------------------------------------------------------------
METHOD: run() - MAIN GAME LOOP
--------------------------------------------------------------------------------

    def run(self):
        """Main game loop"""
        while self.running:
            self.handle_events()
            self.update()
            self.draw()
            self.clock.tick(FPS)

EXPLANATION:
- Runs continuously until self.running = False
- Order matters: events → update → draw
- .tick(60) = limit to 60 FPS (wait if too fast)

GAME LOOP CYCLE:
1. Handle events (input)
2. Update game state (logic)
3. Draw everything (graphics)
4. Wait for next frame
5. Repeat!

        self.game_over()

EXPLANATION:
- Runs when loop exits
- Shows final statistics

--------------------------------------------------------------------------------
METHOD: game_over()
--------------------------------------------------------------------------------

    def game_over(self):
        """Display game over summary"""
        print("\n" + "=" * 60)
        print("🎮 GAME OVER - FINAL STATS")
        print("=" * 60)
        print(f"🏆 Final Score: {self.score}")
        print(f"📊 Total Apples Collected: {self.total_collected}")
        print(f"✅ Good Apples: {self.good_apples}")
        print(f"❌ Damaged Apples: {self.damaged_apples}")

EXPLANATION:
- Prints summary to console
- "=" * 60 = 60 equal signs (divider line)

        if self.total_collected > 0:
            accuracy = (self.good_apples / self.total_collected) * 100
            print(f"📈 Good Apple Rate: {accuracy:.1f}%")

EXPLANATION:
- Calculate percentage of good apples
- .1f = 1 decimal place
- Only if total > 0 (avoid division by zero!)

        pygame.quit()
        sys.exit()

EXPLANATION:
- pygame.quit() = cleanup Pygame resources
- sys.exit() = exit Python program
